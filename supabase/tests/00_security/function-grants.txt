# Every EXECUTE grant on a function in the `public` and `private` schemas,
# other than the owner's own, one per line:
#
#   <grantee> <schema>.<name>(<argument types>)
#
# e.g.  authenticated public.task_capabilities(uuid[])
#       service_role public.submit_application(text, uuid)
#
# The grantee is a role name, or `public` for PUBLIC. Argument types are
# written as Postgres prints them (`oidvectortypes`): comma plus space between
# types, no argument names. Lines starting with `#` and blank lines are
# ignored; order doesn't matter.
#
# supabase/tests/00_security/catalog.test.sql compares this list with the
# database and prints the extra and missing lines when they differ. Add or
# change a line in the same change as the migration that grants it.

# members_members: the helpers RLS policies call (AD-5)
authenticated private.current_member_id()
authenticated private.is_active_member()
authenticated private.is_admin()
authenticated private.is_top_editor()

# tasks_tasks: the only way tasks, task_assignments and activity are written
authenticated public.create_task(text, text, text, text, timestamp with time zone, text, jsonb)

# tasks_responses_and_moves: the transition helpers (called from RLS and from
# task_capabilities, which runs with the caller's rights), the two commands,
# and the one source of offered actions
authenticated private.is_task_approver(uuid)
authenticated private.can_respond_to_slot(uuid)
authenticated private.can_move_task(uuid, task_column)
authenticated public.respond_to_slot(uuid, slot_state, text)
authenticated public.move_task(uuid, task_column)
authenticated public.task_capabilities(uuid[])

# tasks_open_slots: the one definition of an open slot, and the viewer's own
authenticated private.open_slots(uuid)
authenticated public.my_open_slots()

# tasks_realtime: who may join the private task-changes channel (called from
# the realtime.messages policy, which runs as the joining caller)
authenticated private.can_join_task_changes()

# tasks_hand_back: the on_it -> needs_reassignment edge and its command
authenticated private.can_hand_back_slot(uuid)
authenticated public.hand_back_slot(uuid, text)

# members_active_helper_grant: the one activeness definition, which
# member_directory calls with the querying role's rights
authenticated private.is_member_active(uuid)

# members_account_states: the before_user_created Auth hook, run by Supabase
# Auth's own role (refuses every new account but Google's)
supabase_auth_admin private.before_user_created(jsonb)

# members_approval: an admin approves a pending member with positions, or
# declines them
authenticated public.approve_member(uuid, text[], text)
authenticated public.decline_member(uuid)

# core_operations: the server's secret-key client starts and ends job runs,
# records server errors, and pings the database for /api/health
# (members_bootstrap_admin's private.bootstrap_admin has no grant at all: only
# the owner runs it)
service_role public.begin_job_run(text)
service_role public.end_job_run(bigint, job_status, jsonb)
service_role public.record_error_event(text, text, text, uuid, text)
service_role public.health_check()

# tasks_approver_review: the review edges (called from task_capabilities,
# which runs with the caller's rights) and the approver's three commands
authenticated private.can_review_task(uuid)
authenticated private.can_reopen_task(uuid)
authenticated public.mark_done(uuid)
authenticated public.send_back_task(uuid, text)
authenticated public.reopen_task(uuid)

# tasks_delete: the approver's soft delete (its helpers and the recreated
# task_capabilities keep the grants listed above)
authenticated public.delete_task(uuid, timestamp with time zone)
